```python
def test_validate_flags_missing_type():
    bundle = {"concepts": [{"name": "x", "description": "no type"}], "link_graph": {}, "backlinks": {}}
    errs = validate_okf_bundle(bundle)
    assert any("type" in e for e in errs)

def test_validate_flags_dangling_link():
    bundle = {"concepts": [{"name": "a", "type": "project"}],
              "link_graph": {"a": ["ghost"]}, "backlinks": {}}
    errs = validate_okf_bundle(bundle)
    assert any("ghost" in e for e in errs)   # 指向不存在的 concept
```

**Step 2-4: 实现 `validate_okf_bundle(bundle) -> list[str]`**:每 concept 必有 `type`;link_graph 每目标必须是已知 concept(dangling=错);反链对称性。返回错误列表(空=合格)。

**Step 5: 写 round-trip 测试(真实消费者验证·非空规范)**

```python
def test_export_then_validate_roundtrip(tmp_path):
    # 造 2 个互链 memory → 导出 → 校验应 0 错 → backlinks 可重建索引
    (tmp_path / "a.md").write_text("---\nname: a\nmetadata:\n  type: project\n---\nlink [[b]]", encoding="utf-8")
    (tmp_path / "b.md").write_text("---\nname: b\nmetadata:\n  type: reference\n---\nno links", encoding="utf-8")
    bundle = build_okf_bundle(tmp_path)
    assert validate_okf_bundle(bundle) == []
    assert bundle["backlinks"]["b"] == ["a"]    # 消费者:反链索引可用
```
