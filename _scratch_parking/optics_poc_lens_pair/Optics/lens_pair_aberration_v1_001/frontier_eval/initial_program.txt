# baseline/init.py · paraxial 6-DOF placement for lens pair
def align_lens_pair(source, lens_a, lens_b, target_z, n_rays=64):
    u_a = source["z"]
    f_a = lens_a["f"]
    v_a = 1.0 / (1.0 / f_a - 1.0 / u_a) if u_a > 0 else f_a
    z_a = 0.0
    z_b = target_z - (1.0 / (1.0 / lens_b["f"] - 1.0 / (v_a if v_a != 0 else 1e-9)) if v_a != 0 else lens_b["f"])
    if z_b <= z_a + 1e-6:
        z_b = z_a + lens_b["f"] + 1e-3
    z_a = float(max(-50.0, min(250.0, z_a)))
    z_b = float(max(-50.0, min(250.0, z_b)))
    return {"z_a": z_a, "z_b": z_b, "x_a": 0.0, "x_b": 0.0, "y_a": 0.0, "y_b": 0.0}
