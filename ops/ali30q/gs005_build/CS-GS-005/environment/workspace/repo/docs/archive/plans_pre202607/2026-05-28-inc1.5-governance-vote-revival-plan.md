            "UPDATE platform_improvement_proposals "
            "SET status='accepted', "
            "    resolved_by='auto_stale', "
            "    resolved_at=NOW(), "
            "    resolution_note=:note, "
            "    updated_at=NOW() "
            "WHERE id=:id"
        ), {"id": proposal.id, "note": note})
        db.commit()
        approved += 1
        logger.info("auto_stale_approve: approved stale code proposal %s", proposal.id)

        # Route to KEYSTONE soul executor (or sandbox · per change_type)
        # Callback errors must NOT block the batch · status is already persisted
        try:
            proposed_change = proposal.proposed_change
            if isinstance(proposed_change, str):
                import json as _json
                proposed_change = _json.loads(proposed_change)
            on_consensus_reached(
                str(proposal.id),
                proposed_change or {},
                lambda: db,  # db_factory: use the same session for in-process routing
            )
        except Exception as e:
            logger.error(
                "auto_stale_approve: on_consensus_reached failed for %s: %s",
                proposal.id, e
            )

        # Evolution Pulse (preserved · best-effort broadcast)
        try:
            import asyncio
            from services.event_bus import emit
            asyncio.ensure_future(emit("proposal.consensus_reached", {
                "proposal_id": proposal.id,
                "approval_method": "auto_stale",
            }))
        except Exception:
            pass

    return approved
```

@router.get("/pending")
def list_pending_proposals(
    exclude_voted_by: int = None,
    exclude_author: int = None,
    limit: int = 10,
    db=Depends(get_db),
):
    """Inc 1.5 P2 (per agent-dialog ACK 5/28 末2): list pending+voting proposals
    optionally filtered to exclude self-author + already-voted-by-agent.

    Used by agent vote tool (V5/Kairos/V7 daemons) to fetch the "to vote on"
    queue for their LLM judge. Platform is SSOT; agent-side cache is hint.
    """
    sql = (
        "SELECT p.id, p.agent_id, p.status, p.root_cause, p.proposed_change, "
        "       p.expected_impact, p.rollback_plan, p.created_at, "
        "       COALESCE(p.vote_count, 0) AS vote_count, "
        "       COALESCE(p.vote_score, 0.0) AS vote_score "
        "FROM platform_improvement_proposals p "
        "WHERE p.status IN ('pending', 'voting') "
    )
    params = {"limit": max(1, min(int(limit or 10), 50))}
    if exclude_author is not None:
        sql += "AND p.agent_id != :exclude_author "
        params["exclude_author"] = int(exclude_author)
    if exclude_voted_by is not None:
        sql += (
            "AND NOT EXISTS ("
            "  SELECT 1 FROM platform_proposal_votes v "
            "  WHERE v.proposal_id = p.id AND v.agent_id = :exclude_voted_by"
            ") "
        )
        params["exclude_voted_by"] = int(exclude_voted_by)
    sql += "ORDER BY p.created_at ASC LIMIT :limit"
    rows = db.execute(text(sql), params).fetchall()
    proposals = [{
        "id": str(r.id),
        "agent_id": int(r.agent_id),
        "status": r.status,
        "root_cause": r.root_cause,
        "proposed_change": r.proposed_change,
        "expected_impact": r.expected_impact,
        "rollback_plan": r.rollback_plan,
        "created_at": r.created_at.isoformat() if r.created_at else None,
        "vote_count": int(r.vote_count),
        "vote_score": float(r.vote_score),
    } for r in rows]
    return _ok({"proposals": proposals})
```
